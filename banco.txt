CREATE TABLE "Aluno" (
    "RA" INT PRIMARY KEY,
    "Nome" VARCHAR(100) NOT NULL,
    "DataNasc" DATE NOT NULL
);

CREATE TABLE "Disciplina" (
    "Código" INT PRIMARY KEY,
    "Nome" VARCHAR(100) NOT NULL,
    "Carga H." INT NOT NULL
);

CREATE TABLE "Especialidade" (
    "Código" INT PRIMARY KEY,
    "Nome" VARCHAR(100) NOT NULL,
    "Humanas" INT NOT NULL -- 0 para Não, 1 para Sim
);

CREATE TABLE "Professor" (
    "Código" INT PRIMARY KEY,
    "Nome" VARCHAR(100) NOT NULL,
    "Cod. Especialidade" INT NOT NULL,
    "Data Formação" DATE NOT NULL,
    "Salário" DECIMAL(10, 2) NOT NULL,
    CONSTRAINT "FK_Professor_Especialidade" FOREIGN KEY ("Cod. Especialidade") REFERENCES "Especialidade"("Código")
);

CREATE TABLE "AlunoDisciplina" (
    "Código" INT PRIMARY KEY,
    "Cód. Aluno" INT NOT NULL,
    "Cód. Disciplina" INT NOT NULL,
    CONSTRAINT "FK_AlunoDisciplina_Aluno" FOREIGN KEY ("Cód. Aluno") REFERENCES "Aluno"("RA"),
    CONSTRAINT "FK_AlunoDisciplina_Disciplina" FOREIGN KEY ("Cód. Disciplina") REFERENCES "Disciplina"("Código")
);

CREATE TABLE "Turma" (
    "Código" INT PRIMARY KEY,
    "Cód. Aluno" INT NOT NULL,
    "Cód. Disciplina" INT NOT NULL,
    "Cód. Professor" INT NOT NULL,
    CONSTRAINT "FK_Turma_Aluno" FOREIGN KEY ("Cód. Aluno") REFERENCES "Aluno"("RA"),
    CONSTRAINT "FK_Turma_Disciplina" FOREIGN KEY ("Cód. Disciplina") REFERENCES "Disciplina"("Código"),
    CONSTRAINT "FK_Turma_Professor" FOREIGN KEY ("Cód. Professor") REFERENCES "Professor"("Código")
);
